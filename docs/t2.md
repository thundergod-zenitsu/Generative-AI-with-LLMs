# 🔍 pipeline/ocr.py

```python
from doctr.models import ocr_predictor

model = ocr_predictor(pretrained=True)

def extract_blocks(image):
    result = model([image])
    blocks = []

    for page in result.pages:
        for block in page.blocks:
            text = " ".join([line.value for line in block.lines])

            blocks.append({
                "text": text.strip(),
                "bbox": block.geometry,
                "type": "unknown"
            })

    return blocks
```

# 🧠 pipeline/classifier.py

```python
import re

def classify_block(block):
    text = block["text"]

    if re.match(r'^\d+(\.\d+)*\s+[A-Z]', text):
        return "heading"

    if len(text) < 60 and text.isupper():
        return "heading"

    if any(word in text.lower() for word in ["shall", "agreement", "party"]):
        return "clause"

    return "paragraph"


def classify_blocks(blocks):
    for b in blocks:
        b["type"] = classify_block(b)
    return blocks
```

# 🌳 pipeline/hierarchy.py

```python
def build_hierarchy(blocks):
    root = {"title": "DOCUMENT", "children": []}
    current_section = None

    for block in blocks:
        if block["type"] == "heading":
            current_section = {
                "title": block["text"],
                "children": []
            }
            root["children"].append(current_section)

        else:
            if current_section:
                current_section["children"].append(block["text"])
            else:
                root["children"].append(block["text"])

    return root
```

# 🔗 pipeline/graph_builder.py

```python
import networkx as nx

def build_graph(blocks):
    G = nx.DiGraph()

    for i, block in enumerate(blocks):
        G.add_node(i, text=block["text"], type=block["type"])

        if i > 0:
            G.add_edge(i - 1, i, relation="next")

    # heading → clause linking
    last_heading = None
    for i, block in enumerate(blocks):
        if block["type"] == "heading":
            last_heading = i
        elif block["type"] == "clause" and last_heading is not None:
            G.add_edge(last_heading, i, relation="contains")

    return G
```

# 🤖 pipeline/llm.py

```python
from openai import OpenAI
from config import OPENAI_API_KEY, MODEL_NAME

client = OpenAI(api_key=OPENAI_API_KEY)

def structure_with_llm(hierarchy):
    prompt = f"""
Convert this into clean structured JSON with proper nesting.
Remove noise and organize sections logically.

INPUT:
{hierarchy}
"""

    response = client.chat.completions.create(
        model=MODEL_NAME,
        messages=[{"role": "user", "content": prompt}],
        temperature=0
    )

    return response.choices[0].message.content
```

# ⚡ utils/parallel.py

```python
from concurrent.futures import ThreadPoolExecutor
from config import MAX_WORKERS

def parallel_map(func, items):
    with ThreadPoolExecutor(max_workers=MAX_WORKERS) as executor:
        results = list(executor.map(func, items))
    return results
```

# 🔄 pipeline/orchestrator.py

```python
from pipeline.pdf_loader import pdf_to_images
from pipeline.ocr import extract_blocks
from pipeline.classifier import classify_blocks
from pipeline.hierarchy import build_hierarchy
from pipeline.graph_builder import build_graph
from pipeline.llm import structure_with_llm
from utils.parallel import parallel_map


def process_page(image):
    blocks = extract_blocks(image)
    blocks = classify_blocks(blocks)
    return blocks


def run_pipeline(pdf_path):
    print("📄 Loading PDF...")
    images = pdf_to_images(pdf_path)

    print("⚡ Processing pages in parallel...")
    results = parallel_map(process_page, images)

    blocks = [b for page in results for b in page]

    print("🌳 Building hierarchy...")
    hierarchy = build_hierarchy(blocks)

    print("🔗 Building graph...")
    graph = build_graph(blocks)

    print("🤖 Structuring with LLM...")
    structured_json = structure_with_llm(hierarchy)

    return {
        "hierarchy": hierarchy,
        "graph_nodes": len(graph.nodes),
        "graph_edges": len(graph.edges),
        "structured_output": structured_json
    }
```
